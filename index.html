Supabase SQL Schema
Run this once in your Supabase SQL Editor to set up the database
-- GradeView: run this in Supabase SQL Editor
-- https://supabase.com → your project → SQL Editor

create table if not exists public.students (
  id            uuid primary key default gen_random_uuid(),
  teacher_id    uuid references auth.users(id) on delete cascade,
  name          text not null,
  section       text,
  teacher       text,
  access_code   text not null unique,
  grades_all    jsonb not null default '{}',
  created_at    timestamptz default now()
);

-- Index for fast parent lookups by code
create index if not exists idx_students_access_code on public.students(access_code);

-- Index for teacher's own students
create index if not exists idx_students_teacher_id on public.students(teacher_id);

-- Row Level Security: teachers only see/edit their own students
alter table public.students enable row level security;

create policy "Teachers manage own students"
  on public.students for all
  using  ( auth.uid() = teacher_id )
  with check ( auth.uid() = teacher_id );

-- Parents can look up by access_code (anon read)
create policy "Public read by access_code"
  on public.students for select
  using ( true );
